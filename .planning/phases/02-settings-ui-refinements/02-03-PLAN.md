**3. Create backend directory validation endpoint.**

In `backend/src/api/routes/system.ts`, add a `handleValidateDirectory` export:

```typescript
export async function handleValidateDirectory(
  _req: IncomingMessage,
  res: ServerResponse,
  _params: Record<string, string>,
  body: unknown
): Promise<void> {
  const { path: dirPath } = body as { path: string };
  if (!dirPath || typeof dirPath !== 'string') {
    // Return 400 { valid: false, error: 'No path provided' }
    return;
  }
  try {
    const stat = fs.statSync(dirPath);
    if (!stat.isDirectory()) {
      // Return 200 { valid: false, error: 'Path is not a directory' }
      return;
    }
    fs.accessSync(dirPath, fs.constants.R_OK | fs.constants.X_OK);
    // Return 200 { valid: true }
  } catch {
    // Return 200 { valid: false, error: 'Directory does not exist or is inaccessible' }
  }
}
```

This endpoint accepts `{ path: string }` in the POST body and returns `{ valid: boolean, error?: string }`. It checks: path exists, is a directory, and is readable+executable (accessible). Always returns 200 with valid/error payload (not 4xx for invalid paths — the path being invalid is a valid response, not an error).

const handleBrowse = async () => {
  setDirError(''); // Clear previous error
  try {
    const res = await fetch('/api/v1/system/pick-directory', { method: 'POST' });
    if (!res.ok) {
      setDirError('Failed to open folder picker');

    }
    const data = await res.json();
    if (data.error) {
      setDirError(data.error);
      return;
    }

      setDefaultStartDir(data.path);
    }
    // If cancelled (data.cancelled), do nothing — no error

  }
};

**Manual path validation on blur/Enter:**
When the user types or pastes a path and presses Enter or blurs the input, validate it against the backend:
```typescript
const handleDirCommit = async (value: string) => {
  setDirError(''); // Clear previous error
  if (!value.trim()) {
    // Empty = OS default, valid
    setDefaultStartDir('');
    return;
  }
  try {
    const res = await fetch('/api/v1/system/validate-directory', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ path: value.trim() }),
    });
    const data = await res.json();
    if (data.valid) {
      setDefaultStartDir(value.trim());
    } else {
      setDirError(data.error || 'Invalid directory path');
      setDefaultStartDir(''); // Fall back to OS default per user decision
    }
  } catch {
    setDirError('Could not validate directory');
    setDefaultStartDir(''); // Fall back to OS default
  }
};
```

Bind `handleDirCommit` to the input's `onBlur` and `onKeyDown` (Enter key) events. Use a local controlled input value so the user can type freely before committing.
